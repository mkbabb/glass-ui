+/* utilities/engage.css — ONE RECIPE PER ENGAGEMENT CLASS.
+ *
+ * `ENGAGE_DECLARATION` (composables/motion/engage/engageClasses.ts) says what each
+ * component directory is; this file says what each class does. A member composes its
+ * class's utility on the element it paints, and every utility stamps its class as the
+ * registered `--rung-class` (tokens/property-regs.css), which the runtime census reads
+ * to prove the member composes the recipe its row declares. Where a class's answer
+ * belongs to a substrate that already paints the member (the field's focus paint, the
+ * sortable row's fill, the modal veil), the substrate keeps it and the utility stamps.
+ *
+ * THE PRESS IS NEVER A CLASS'S GEOMETRY. A member presses in LIGHT on its own paint,
+ * or by script where the engage door owns the press (`useEngage`: the Button's squish
+ * on the one `--press-t` drive). No recipe here writes `scale` for a press, and none
+ * sets `scale: none` to silence another class: the press-snap and hover lift that used
+ * to ride `.tap-squish`, `.interactive-item` and `.glass-capsule-hover` are retired at
+ * their source, not overridden here.
+ *
+ * THE COMMIT IS LIGHT (CK-02 arm iii, Q-5's default). On the commit edge the painted
+ * face brightens and the light drains back to the committed rest, on the
+ * `control-engage` envelope ("the engagement wash and bloom on a control surface"):
+ * a GATE opens on the row's attack while a CHARGE drains on its release, and the light
+ * is their product. The two states swap which leg is fast, so the product rises and
+ * falls on EVERY edge, in either direction, from transitions alone:
+ *   · committed  (state on, open, or the parity "b"): the gate opens (opacity 0 → 1,
+ *     attack) over a charge that drains (light → none, release);
+ *   · released   (state off, closed, or the parity "a"): the charge returns (none →
+ *     light, attack) under a gate that closes (opacity 1 → 0, release).
+ * At rest in either state the product is zero, so a member that MOUNTS committed shows
+ * nothing (a transition needs a before-change style, and an inserted element has none),
+ * a commit from the model and one from a gesture take one path, and a second commit
+ * mid-flight retargets both legs. It is light, so it stays under reduced motion. Its
+ * seat is the member's PAINTED FACE: the element's own `::after`, under its content
+ * (`z-index: -1` in its own stacking context, so a label never washes out), or, on a
+ * face that carries a mark (the control bit's thumb, tick or dot), the face's last
+ * layer `[data-engage-light]`, over the mark, so the whole 20 px face brightens and
+ * not only the rim the mark leaves showing.
+ *
+ * Clocks: geometry on the spring table's generated tokens, light on the envelope
+ * table's (`--engage-*`). The commit's amplitude is the fill ladder's state rung
+ * (`--fill-selected`); its light is the page's own ground (`--neutral-0` in light,
+ * the ink in dark: the brighter of the two in each scheme). */
+
+@property --rung-grow {
+    syntax: "<number>";
+    inherits: false;
+    initial-value: 0;
+}

+/* PRESSED — an action with no state (button, action chip). The press is the member's
+   light or the door's script press; the commit edge is the door's parity
+   (`data-commit` "a" ↔ "b", `useEngage` `commit: "call"`). */
+@utility engage-pressed {
+    --rung-class: pressed;
+}

+/* CHOSEN — a control with a committed state (switch, checkbox, radio, selectable chip,
+   toggle item, segmented tabs). The press is LIGHT on the member's own paint; the
+   commit is the light below, or, where the member has a travelling body (the tabs'
+   indicator), that body's travel on its own row. */
+@utility engage-chosen {
+    --rung-class: chosen;
+}

+/* DISCLOSED — a trigger that opens a region (accordion, collapsible, popover, select,
+   menu). A row-width target answers a press in LIGHT only (a full-width squish reads
+   as the row collapsing), on the press envelope's attack; it commits in light when the
+   region opens and when it closes. Its hover is the substrate's
+   (components/_shared/disclosure/disclosure.css), under `(hover: hover)`.
+   The press is a WASH, not a slab: a press is a transient pointer answer, which is the
+   fill ladder's `--fill-hover` rung (tokens/color-radius.css §1.3); `--fill-selected` is
+   persistent state, and at 12 % across a 320 px row it read as an opaque grey bar
+   (D5 pass 3 C-A3 §3 item 2). It keys on the door's press mark (`data-press`), which
+   lands on `pointerdown`, not on `:active` (composables/motion/engage/useEngage.ts). */
+@utility engage-disclosed {
+    --rung-class: disclosed;
+    transition:
+        background-color var(--engage-press-drain-release) var(--ease-standard),
+        color var(--duration-control) var(--ease-standard),
+        box-shadow var(--duration-control) var(--ease-standard);
+
+    &[data-press]:not(:disabled, [data-disabled]) {
+        background-color: color-mix(
+            in oklab,
+            var(--foreground) calc(var(--fill-hover) * 100%),
+            transparent

+        transition-duration:
+            var(--engage-press-drain-attack), var(--duration-control),
+            var(--duration-control);
+    }
+
+    /* FL-3: under forced colours the system palette owns the fill, so the wash would
+       compute Canvas and the press would vanish; the press is the system's Highlight
+       ring instead, the disclosure's own focus idiom (disclosure.css). Under reduced
+       transparency the wash stays: it is an opaque-ink rung, not a material. */

+            outline: 2px solid Highlight;
+            outline-offset: -2px;
+        }

+/* HELD — a control the hand holds (slider, number-field scrub, timeline, the dock's
+   controls). The ENGAGED rung in place on a fine pointer: the host grows to
+   `--scale-engage` on the `dock` row while the door marks it engaged (`data-engaged`:
+   the hand is down, or a key brought the focus or acts on it), and leaves on the row's
+   exit clock. On a coarse pointer it does not grow in place: the rung there is a
+   promotion this class does not own (Q-3 (c)). */
+@utility engage-held {
+    --rung-class: held;
+    scale: calc(1 + (var(--scale-engage) - 1) * var(--rung-grow));
+    transform-origin: center;
+    /* the host's ink leg rides with the grow: the slider lane (unlayered) declares no
+       transition of its own, so this list is the host's one */
+    transition:

+        color var(--duration-fast) var(--ease-standard);
+
+    @media (hover: hover) and (pointer: fine) {

+            transition-duration: var(--spring-dock-duration), var(--duration-fast);

+    @media (prefers-reduced-motion: reduce) {
+        scale: none;
+    }
+
+    /* FL-3: the grow is geometry, which neither forced colours nor reduced transparency

+/* PRESENTED — a surface raised over the page (dialog, sheet, toast, tooltip). Its rung
